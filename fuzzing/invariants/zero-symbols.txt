# App-local symbols to zero out in the invariant (merged with the SDK's
# default zero list). These globals are pure scratch / display buffers: the
# app reads back from them only after writing, so leaving them fuzzable
# would just inflate the prefix without unlocking new branches.

# Action serialization scratch buffer (1024 bytes).
g_raw_buf@action.c

# NBGL display formatting buffer; reset on every UI flow.
g_ui_ctx@operation.c

# g_signing_ctx@sign_action.c is intentionally NOT zeroed: the non-first
# action branch in handler_sign_action() does memcmp(tmp, g_signing_ctx)
# before calling sign_action(). When this global is zeroed and the APDU
# carries a non-empty BIP32 path, the memcmp always fails and sign_action()
# (along with action_hash, eip712_*_hash, eip712_sign, action_serialize, and
# the msgpack writers in src_msgpack/cmp.c) is unreachable. Leaving it
# fuzzable lets seeds pre-populate a path that matches the APDU payload,
# unlocking ~185 lines of EIP-712 code and ~150+ lines of msgpack writers
# behind action_hash. Cost: 44 extra prefix bytes per input.
